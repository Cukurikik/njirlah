## 🔧 API ROUTES (Next.js App Router)

### /api/chat  
POST handler using Vercel AI SDK streamText:
```typescript
import { streamText } from 'ai'
import { createOpenRouter } from '@openrouter/ai-sdk-provider'

export async function POST(req: Request) {
  const { messages, model, apiKey } = await req.json()
  
  const openrouter = createOpenRouter({ apiKey })
  
  const result = await streamText({
    model: openrouter(model),
    messages,
    system: `You are NJIRLAH AI, an expert coding assistant...`
  })
  
  return result.toDataStreamResponse()
}
```

### /api/cloudflare
POST handler for Cloudflare Workers AI:
```typescript
export async function POST(req: Request) {
  const { prompt, model, accountId, apiToken } = await req.json()
  
  const response = await fetch(
    `https://api.cloudflare.com/client/v4/accounts/${accountId}/ai/run/${model}`,
    {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${apiToken}` },
      body: JSON.stringify({ prompt })
    }
  )
  
  return Response.json(await response.json())
}
```
